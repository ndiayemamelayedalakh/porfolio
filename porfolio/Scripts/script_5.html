# ----------------------  Importation des données ----------------------

#Charger les données à partir d’un fichier RDS
data = readRDS(file = "Data/Accidents.rds")

# Afficher la structure du jeu de données (types de colonnes, aperçu des valeurs)
str(data)

#Afficher des statistiques descriptives globales
summary(data)

# ----------------------  Transformation de colonnes ----------------------

#Renommer certaines colonnes et convertir des types
data <- within(data, {
  Lumière <- lum       # Renommer la variable 'lum' en 'Lumière' (plus lisible)
  lum <- NULL          # Supprimer l'ancienne colonne 'lum'
  mois <- factor(mois) # Transformer la variable 'mois' en facteur
})

# ----------------------  Filtrage des départements normands ----------------------

#Garder uniquement les lignes correspondant aux départements de la Normandie
normandie <- subset(data, dep %in% c(14, 27, 50, 61, 76))

# Remplacer les codes des départements par leurs noms
normandie$dep <- factor(normandie$dep,
                        levels = c(14, 27, 50, 61, 76),
                        labels = c("Calvados", "Eure", "Manche", "Orne", "Seine-Maritime"))

#----------------------  Nettoyage des valeurs manquantes ----------------------

#Remplacer la modalité -1 (valeur invalide) par NA dans les colonnes 'grav' et 'sexe',
#puis convertir ces variables en facteurs
normandie <- within(normandie, {
  grav[grav == -1] <- NA     # Valeurs invalides en NA
  grav <- factor(grav)       # Conversion en facteur
  sexe[sexe == -1] <- NA
  sexe <- factor(sexe)
})

#----------------------  Création d’une variable dérivée ----------------------

# Calculer l’âge de la victime au moment de l’accident (année accident - année naissance)
normandie$age <- as.numeric(normandie$Year - normandie$an_nais)

#----------------------  Suppression des doublons ----------------------

#Identifier les doublons
duplicated(normandie)

#Supprimer les lignes dupliquées
normandie <- normandie[!duplicated(normandie), ]

#----------------------  Vérification finale ----------------------

#Statistiques descriptives finales après nettoyage
summary(normandie)

#Structure finale du dataframe
str(normandie)
summary(normandie$Year)

#Installation de la librairaie lsr
install.packages("lsr")

#Graphique
for (horaire in rownames(tab)) {
  effectifs <- tab[horaire, ]
  
  graph11 <- barplot(effectifs,
                     names.arg = colnames(tab),
                     col = c("#b82020","#1ce4e4","#7d827f","#7abb79","#d2d2d2"),
                     ylim = c(0, max(tab) + 5),
                     main = paste("Tranche horaire :", horaire),
                     xlab = "Département",
                     ylab = "Nombre de tués",
                     cex.names = 1)
  
#Etiquettes
  text(x = graph11,
       y = effectifs + 2,
       labels = effectifs,
       cex = 0.9)
}

#Titre principal
mtext("Effectifs de tués selon le département pour chaque tranche horaire", 
      outer = TRUE, line = 5.5, cex = 1.4, font = 2)

#Légende
par(fig = c(0, 1, 0.97, 1), new = TRUE, mar = c(0, 0, 0, 0))  
plot.new()
legend("center",
       legend = colnames(tab),
       fill = c("#b82020","#1ce4e4","#7d827f","#7abb79","#d2d2d2"),
       horiz = TRUE,
       bty = "n",
       cex = 1)

#Test du chi² d’indépendance
chi_squared <- chisq.test(tab)
print(chi_squared)

#Calcul du V de Cramer
n_total <- sum(tab)
k <- min(nrow(tab), ncol(tab)) 
V_cramer <- sqrt(chi_squared$statistic / (n_total * (k - 1)))

# Affichage de la valeur du V de Cramer
cat("Valeur du V de Cramer :", round(V_cramer, 3), "\n")

conducteur <- subset(data_grav,
                     catu == 1 & grav %in% c("Tué", "Blessé hospitalisé"))

# Diagramme : nuage de dispersion et courbe de régression lissée
with(conducteur,
     loess(age ~ heure, span = 0.8)) -> smooth

with(conducteur,
     plot(age ~ heure,
          pch = 21,
          col = "black",
          bg = "blue",
          las = 1,
          cex = 0.7,
          main = "Nuage de dispersion : age versus heure",
          xlab = "Variable heure",
          ylab = "Variable age"))

with(conducteur,
     lines(fitted(smooth) ~ heure,
           col = "red",
           lwd = 3))

smooth = data.frame(heure = conducteur$heure,
                    smooth = fitted(smooth))
smooth = smooth[order(smooth$heure,decreasing = FALSE),]

#Mettre les 2 graphiques en matrices
par(mfrow = c(1, 2))

#Filtrer les conducteurs uniquement (catégorie 1)
conducteur = subset(normandie,
                    subset = catu == '1',
                    select = c(age, grav))

#Remplacer les codes par les libellés
conducteur$grav <- factor(conducteur$grav,
                           levels = c(1, 2, 3, 4),
                           labels = c("Indemne", "Tué", "Blessé hospitalisé", "Blessé léger"))
conducteur

#Calcul des moyennes d'âge par type de gravité
aggregate(x = conducteur$age,
          by = list(Gravite = conducteur$grav),
          FUN = mean,
          na.rm = TRUE) -> mean.age

#Affichage de la moyenne
mean.age

#Graphique
with(data = conducteur,
     expr = boxplot(age ~ grav,
            col = c("lightblue", "red", "red4", "blue"),           
            main = "Répartition de l'âge des conducteurs selon \n la gravité de l'accident",
            xlab = "Gravité",
            ylab = "Age des conducteurs"))

#Affichage d'un boxplot de l'âge selon la gravité
points(x = 1:4,
       y = mean.age$x,
       pch = 20,
       cex = 1.5,
       col = "black")

#Relier les points des moyennes avec une ligne
lines(x = 1:4,
      y = mean.age$x,

      lwd = 2,        
      lty = 1)

                probability = TRUE,

                main = "Répartition des accidentés tués selon l'âge en Normandie en 2023",
                xlab = "Âge des tués",
                ylab = "Densité")

#Ajout des densités sur chaque barre
text(x = graph15$mids,
     y = graph15$density+0.001,
     labels = graph15$counts,
     cex = 0.8)

#Moyenne
abline(v = mean(Deces_nor_2023$age),
       lty = 2,
       col = "blue2",
       lwd = 2)
